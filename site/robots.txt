# Everything here is public book metadata and meant to be found.
#
# The catalogue itself is a single-page app routed on the URL hash, which is
# not an address a crawler can index. So the browsable scopes are also
# published as ordinary pages — one per publisher, category, dataset, year
# and language — generated at publish time by tools/build-pages.js and listed
# in the sitemap. /browse/ is the index of them, and the app's footer links
# to it, so none of them depends on the sitemap alone to be found.

User-agent: *
Allow: /

# A developer page for inspecting the database connection. No catalogue
# content, nothing a reader wants in a search result.
Disallow: /diagnose.html
